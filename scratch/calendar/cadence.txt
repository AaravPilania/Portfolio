new Promise((res) => {
    const gaps = [], seam = []; let last = null, lastT = 0, prevT = null, prevKey = '';
    const end = performance.now() + 19000;
    function f(ts) {
        const t = __calendarContact.time(), k = __calendarContact.scene(t);
        const m = k.match(/^dance@\d+x\d:(\d+)/);
        if (m) {
            if (last !== null && m[1] !== last) { gaps.push(ts - lastT); lastT = ts; }
            if (last === null) lastT = ts;
            last = m[1];
        } else last = null;
        if (prevT !== null && t < prevT - 8) seam.push([+prevT.toFixed(3), prevKey, +t.toFixed(3), k]);
        prevT = t; prevKey = k;
        if (ts < end) requestAnimationFrame(f);
        else {
            const vs = 1000 / 60, hist = {};
            gaps.forEach((d) => { const b = Math.round(d / vs); hist[b] = (hist[b] || 0) + 1; });
            res({ danceSteps: gaps.length, stepsInVsyncs: hist, seam });
        }
    }
    requestAnimationFrame(f);
})
